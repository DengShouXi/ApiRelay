### 2. Agent 必须亲自跑的检查命令（补漏前 + 补漏后各一遍）

在 `ApiRelay/` 工程目录下执行（或把路径换成绝对路径）：

```bash
# 2a. Catalog：sourceLanguage、缺口、垃圾 key
python3 - <<'PY'
import json
from pathlib import Path
p = Path("ApiRelay/Localizable.xcstrings")  # 若 cwd=ApiRelay/ApiRelay 则改 Path("Localizable.xcstrings")
data = json.loads(p.read_text())
print("sourceLanguage=", data.get("sourceLanguage"))
gaps = []
for k, v in data["strings"].items():
    locs = v.get("localizations") or {}
    en = ((locs.get("en") or {}).get("stringUnit") or {}).get("value")
    zh = ((locs.get("zh-Hans") or {}).get("stringUnit") or {}).get("value")
    if not en or not zh:
        gaps.append((k, bool(en), bool(zh)))
print("gap_count=", len(gaps))
for row in gaps[:50]:
    print(row)
junk = [k for k in data["strings"] if k in ("%lld", "••••%@", "%@，%@") or (k.startswith("%") and "error." not in k and "vault." not in k and "settings." not in k)]
print("suspect_junk=", junk)
PY
```

```bash
# 2b. 源码：面向用户的硬编码中文（注释除外）
python3 - <<'PY'
import re
from pathlib import Path
root = Path("ApiRelay")  # 含 UI/Business 的源码根；按 cwd 调整
hits = []
for p in root.rglob("*.swift"):
    for i, line in enumerate(p.read_text(errors="ignore").splitlines(), 1):
        s = line.strip()
        if s.startswith("//") or s.startswith("///"):
            continue
        code = re.sub(r"//.*$", "", line)
        if any("\u4e00" <= c <= "\u9fff" for c in code):
            hits.append(f"{p}:{i}: {s[:120]}")
print("cjk_user_facing=", len(hits))
for h in hits:
    print(h)
PY
```
